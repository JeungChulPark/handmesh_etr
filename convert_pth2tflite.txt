### Install openvino ###
wget https://registrationcenter-download.intel.com/akdlm/irc_nas/18319/l_openvino_toolkit_p_2021.4.752.tgz
tar -xvzf l_openvino_toolkit_p_2021.4.752.tgz
cd l_openvino_toolkit_p_2021.4.752
chmod +x install.sh
./install.sh (select all 1)
sudo -E ./install_openvino_dependencies.sh
pip install openvino2tensorflow --upgrade
pip install onnxruntime --upgrade
pip install onnx-simplifier --upgrade
pip install defusedxml
pip install tensorflow-datasets
### pth → onnx ###
python convert_to_onnx.py --resolution [240, 240] --name model
### onnx optimize ###
python3 -m onnxsim model.onnx model_opt.onnx
### onnx → openvino ###
python3 /opt/intel/openvino_2021/deployment_tools/model_optimizer/mo.py --input_model model_opt.onnx --input_shape [1,3,240,240] --output_dir model --data_type FP32
### openvino → tf, tflite ###
openvino2tensorflow --model_path model/model_opt.xml --model_output_path model/saved_model --output_saved_model --output_h5 --output_pb --output_no_quant_float32_tflite --output_weight_quant_tflite --output_float16_quant_tflite
result : model → saved_model → model_float32.tflite
